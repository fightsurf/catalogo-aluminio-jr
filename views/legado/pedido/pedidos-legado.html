function marcarResultadoAtivo(indexAtivo) { document.querySelectorAll('.resultado-item').forEach((item) => { item.classList.toggle('ativo', Number(item.dataset.index) === Number(indexAtivo)); }); pedidosEncontrados = pedidosEncontrados.filter((_, index) => index === Number(indexAtivo)); resultadoLista.innerHTML = pedidosEncontrados.map((pedido, index) => `
Pedido ${pedido?.numero || '-'} · ${pedido?.cliente?.nome || '-'}
Data: ${formatarData(pedido?.data)} · Cidade: ${pedido?.cliente?.cidade || '-'} / ${pedido?.cliente?.uf || '-'} · Total: ${moeda(pedido?.total || 0)}
`).join(''); }
